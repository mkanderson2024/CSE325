// Generate a formated sales summarry report string
string GenerateSalesSummaryReport(IEnumerable<string> salesFiles)
{
    var summary = new StringBuilder();
    double summaryTotal = 0;
    var fileDetails = new List<(string storeId, double Total)>();

    summary.AppendLine("============= Sales Summary =============");

    foreach (var file in salesFiles)
    {
        string salesJson = File.ReadAllText(file);
        SalesData? data = JsonConvert.DeserializeObject<SalesData?>(salesJson);
        double fileTotal = data?.Total ?? 0;

        summaryTotal += fileTotal;

        var storeId = Path.GetFileName(Path.GetDirectoryName(file));

        fileDetails.Add((storeId ?? "Unknown", fileTotal));
    }

    summary.AppendLine($"Total Sales: {summaryTotal:C}");
    summary.AppendLine();
    summary.AppendLine("Details");

    foreach (var detail in fileDetails)
    {
        summary.AppendLine($"  {detail.storeId}: {detail.Total:C}");
    }
    return summary.ToString();
}
